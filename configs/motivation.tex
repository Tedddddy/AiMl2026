\begin{frame}{Motivation}
    \begin{itemize}
        \item Product of Kripke frames were studied by Shehtman and Segerberg  in 70s\pause
        \item $\Box_1 \Box_2p \leftrightarrow \Box_1\Box_2p$ (com) and $\lozenge_1 \Box_2 p \rightarrow \Box_2 \lozenge_1 p$ (chr) are valid \pause
        \item $S4 \times S4 = S4 \otimes S4 + $ (com) $+$ (chr) \pause 
        \item (com) and (chr) fails on products of topological frames \pause
        \item $S4 \times_t S4 = S4 \otimes S4$ \pause
        \item $S4 \times_t^+ S4 = S4 \otimes S4 \otimes S4 + \Box p \to \Box_1p \land \Box_2p$ 
    \end{itemize}
\end{frame}
