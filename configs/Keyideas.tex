\begin{frame}{Main Theorem}
    \begin{theorem}
        Let $\Lambda$ be a unimodal logic with $\Lambda \in \{T,D\}$. Then the following holds:
        \[
            \Lambda \otimes \Lambda \otimes \Lambda + \Box p \to \Box_1\Box_2p \land \Box_2\Box_1p = \Lambda \times_n^+ \Lambda
        \]
    \end{theorem}
    In the following we will sketch the proof for $\Lambda = T$.

        \vspace{1.5cm}
         We denote $T \otimes T \otimes T + \Box p \to \Box_1\Box_2p \land \Box_2\Box_1p$ as $T_3m$

\end{frame}
